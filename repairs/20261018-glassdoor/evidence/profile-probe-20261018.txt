```python
"""Sonda acotada (2026-10-18, fase de investigacion repair/glassdoor-20261018)."""
from __future__ import annotations

import time

from src.core.browser import launch_context, close_context

try:
    from patchright.sync_api import sync_playwright
except ImportError:
    from playwright.sync_api import sync_playwright  # type: ignore


def main() -> int:
    t0 = time.time()
    pw = sync_playwright().start()
    context = None
    try:
        context = launch_context(
            pw, locale="en-IE", timezone="Europe/Dublin",
            profile_name="glassdoor")
        print("launch=OK seconds=%.1f" % (time.time() - t0), flush=True)
        page = context.new_page()
        t1 = time.time()
        try:
            page.goto("https://www.glassdoor.ie/", wait_until="domcontentloaded",
                      timeout=30000)
            print("goto=OK seconds=%.1f title=%r url=%r"
                  % (time.time() - t1, page.title(), page.url), flush=True)
        except Exception as e:
            print("goto=FAIL seconds=%.1f error=%s" % (time.time() - t1, e),
                  flush=True)
        t2 = time.time()
        try:
            page.goto("https://www.glassdoor.ie/Job/jobs.htm?sc.keyword=Data+Analyst&fromAge=7",
                      wait_until="domcontentloaded", timeout=30000)
            print("serp_goto=OK seconds=%.1f title=%r url=%r"
                  % (time.time() - t2, page.title(), page.url), flush=True)
        except Exception as e:
            print("serp_goto=FAIL seconds=%.1f error=%s" % (time.time() - t2, e),
                  flush=True)
        try:
            page.close()
        except Exception:
            pass
        return 0
    finally:
        if context is not None:
            close_context(context)
        try:
            pw.stop()
        except Exception:
            pass


if __name__ == "__main__":
    raise SystemExit(main())
```
